```rust
#[test]
fn issue_path_sort() {
    // A smaller, distinct reproduction of the public --sort path request.
    let root = tempfile::tempdir().expect("create public input directory");
    for name in ["z.txt", "m.txt", "a.txt"] {
        std::fs::write(root.path().join(name), b"public input").expect("write input");
    }
    let binary = env!("CARGO_BIN_EXE_fd");
    let version = std::process::Command::new(binary).arg("--version")
        .output().expect("run candidate-built fd");
    assert!(version.status.success(), "fd preflight failed: {:?}", version);
    let output = std::process::Command::new(binary).current_dir(root.path())
        .args(["--color", "never", "--type", "f", "--sort", "path", "."])
        .output().expect("run fd sort request");
    assert!(output.status.success(), "--sort path must be accepted: {:?}", output);
    let stdout = String::from_utf8(output.stdout).expect("UTF-8 public output");
    let actual: Vec<_> = stdout.lines().map(|s| s.trim_start_matches("./")).collect();
    assert_eq!(actual, vec!["a.txt", "m.txt", "z.txt"]);
}
```
